local function freeze()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	char:WaitForChild("Humanoid").PlatformStand = true
	task.spawn(function()
		for _, p in ipairs(char:GetDescendants()) do
			if p:IsA("BasePart") and not p.Anchored then p.Anchored = true end
		end
	end)
end

local function unfreeze()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	char:WaitForChild("Humanoid").PlatformStand = false
	task.spawn(function()
		for _, p in ipairs(char:GetDescendants()) do
			if p:IsA("BasePart") and p.Anchored then p.Anchored = false end
		end
	end)
end

local function refresh()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	char:WaitForChild("Humanoid"):ChangeState(Enum.HumanoidStateType.Freefall)
end

local function refreshSwim()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local hum = char:WaitForChild("Humanoid")
	hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	task.wait(0.1)
	hum:ChangeState(Enum.HumanoidStateType.Swimming)
end

local function refreshClimb()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local hum = char:WaitForChild("Humanoid")
	hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	task.wait(0.1)
	hum:ChangeState(Enum.HumanoidStateType.Climbing)
end

function setAnimation(animType, animId)
	if type(animId) ~= "table" and type(animId) ~= "string" then return end
	local char = LocalPlayer.Character
	if not char then return end
	local Animate = char:FindFirstChild("Animate")
	if not Animate then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end

	freeze()
	task.wait(0.1)
	for _, track in ipairs(hum:GetPlayingAnimationTracks()) do track:Stop(0) end

	local ok, err = pcall(function()
		pcall(resetFns[animType], Animate)
		lastAnimations[animType] = animId
		applyFns[animType](Animate, animId)
	end)
	if not ok then warn("setAnimation error:", err) end

	pcall(function() writefile("MeWhenUrMom.json", HttpService:JSONEncode(lastAnimations)) end)
	task.wait(0.1)
	unfreeze()
end

-- Reset character + keep exact position
local function fullReset()
	local char = LocalPlayer.Character
	if not char then return end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	local savedCFrame = hrp and hrp.CFrame or CFrame.new()

	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.Health = 0
	end

	local conn
	conn = LocalPlayer.CharacterAdded:Connect(function(newChar)
		conn:Disconnect()
		task.wait(0.15)
		local newHrp = newChar:WaitForChild("HumanoidRootPart", 3)
		if newHrp then
			newChar:PivotTo(savedCFrame)
		end
	end)
end

local function loadLastAnimations()
	if isfile("MeWhenUrMom.json") then
		local ok, data = pcall(readfile, "MeWhenUrMom.json")
		if ok then
			local saved = HttpService:JSONDecode(data)
			for _, t in ipairs({"Idle","Walk","Run","Jump","Fall","Swim","SwimIdle","Climb"}) do
				if saved[t] then setAnimation(t, saved[t]) end
			end
		end
	end
end

LocalPlayer.CharacterAdded:Connect(function(char)
	char:WaitForChild("Humanoid")
	char:WaitForChild("Animate", 10)
	task.wait(0.3)
	for _, t in ipairs({"Idle","Walk","Run","Jump","Fall","Swim","SwimIdle","Climb"}) do
		if lastAnimations[t] then pcall(setAnimation, t, lastAnimations[t]) end
	end
end)

local function isCurrentAnim(animType, id)
	local last = lastAnimations[animType]
	if not last then return false end
	if type(last) == "table" and type(id) == "table" then
		return tostring(last[1]) == tostring(id[1])
	end
	return tostring(last) == tostring(id)
end

	for n in pairs(typeAnims) do table.insert(names, n) end

	for _, name in ipairs(names) do

	for _, t in ipairs(animTypes) do

			if t == typeName then
				tween(btn, {BackgroundColor3 = THEME.Accent}, HOVER_TWEEN_INFO)
				btn.TextColor3 = Color3.fromRGB(255, 255, 255)
			else
				tween(btn, {BackgroundColor3 = THEME.Panel}, HOVER_TWEEN_INFO)
				btn.TextColor3 = THEME.TextMuted
			end
		end
	end
	if tabBtns["Custom"] then
		if typeName == "Custom" then
			tween(tabBtns["Custom"], {BackgroundColor3 = THEME.Accent}, HOVER_TWEEN_INFO)

			tabBtns["Custom"].TextColor3 = THEME.TextMuted
		end
	end

	if not isCustom then
		populate(typeName, SearchBox.Text)
	end
end

local function updateDrag(input)
	local delta = input.Position - dragStart
	frame.Position = UDim2.new(
		startPos.X.Scale, startPos.X.Offset + delta.X,
		startPos.Y.Scale, startPos.Y.Offset + delta.Y
	)

	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = frame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)
	end
end)

	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging and input == dragInput then updateDrag(input) end
end)

local function LoadAndUpdateAnimations()
	local ok, result = pcall(function()
		return game:HttpGet("https://raw.githubusercontent.com/Gazer-Ha/Gaze-stuff/refs/heads/main/Gaze%20Anim%20Database")
	end)
	if not ok or not result then return end

	local fn = loadstring(result)
	if not fn then return end
	local pok, onlineAnims = pcall(fn)
	if not pok or type(onlineAnims) ~= "table" then return end

	local changed = false

	local function sameAnim(a, b)
		if type(a) ~= type(b) then return false end
		if type(a) == "table" then
			if #a ~= #b then return false end
			for i = 1, #a do
				if tostring(a[i]) ~= tostring(b[i]) then return false end
			end
			return true
		end
		return tostring(a) == tostring(b)
	end

	for animType, typeAnims in pairs(onlineAnims) do
		local fileType = fileAnimations[animType] or {}
		for name, data in pairs(typeAnims) do
			if not fileType[name] then
				Animations[animType] = Animations[animType] or {}
				Animations[animType][name] = data
				fileAnimations[animType] = fileAnimations[animType] or {}
				fileAnimations[animType][name] = data
				changed = true
			elseif not sameAnim(data, fileType[name]) then
				Animations[animType][name] = data
				fileAnimations[animType][name] = data
				changed = true
			end
		end
	end

	if changed then
		pcall(writefile, "GreyLikesToSmellUrFeet.json", HttpService:JSONEncode(Animations))
		populate(currentTab, SearchBox.Text)
	end
end

frame.Size = UDim2.new(0, 340, 0, 0)
tween(frame, { Size = OPEN_SIZE })

setTab("Idle")
loadLastAnimations()

task.delay(1, function()
	LoadAndUpdateAnimations()
end)

end)
